```tsx
  AppSwitcher,
  type AppSwitcherProps,

  type AppSwitcherApp,
} from '@innostes/ui';
```

## When to use

Use `AppSwitcher` in the top application bar to enable direct transitions between major ERP business systems:

- **Ecosystem module navigation**: Jumping between Procurement, Warehousing, Invoicing, and Human Resources.
- **Multi-tenant portal switching**: Switching between Northwind Traders distribution entities.
- **Developer tools and settings**: Moving from core operational views to analytics consoles.

**Do not use AppSwitcher for:**
- Page-level sub-navigation within a single module — use [Sidebar](/ui/docs/components/layout/sidebar).
- Flat icon navigation rails — use [AppRail](/ui/docs/components/layout/app-rail).
- Standalone command execution — use [CommandPalette](/ui/docs/components/overlays/command-palette).

## Examples

### Grouped modules with search filter

Organising business modules under titled sections with built-in search:

```tsx
const sections: AppSwitcherSection[] = [
  {
    label: 'Supply chain & operations',
    apps: [
      {
        id: 'procurement',
        name: 'Procurement',
        description: 'Purchase orders, vendors, and RFQs',
        icon: <ShoppingCart className="size-5" />,
        tone: 'chart-1',
        href: '/procurement',
      },
      {
        id: 'inventory',
        name: 'Warehousing & inventory',
        description: 'Stock management, bins, and e-way bills',
        icon: <Package className="size-5" />,
        tone: 'chart-2',
        href: '/inventory',
      },
    ],
  },
  {
    label: 'Finance & compliance',
    apps: [
      {
        id: 'billing',
        name: 'Invoices & GST',
        description: 'Tax invoices, credit notes, and GSTR-1',
        icon: <FileText className="size-5" />,
        tone: 'chart-3',
        href: '/finance',
      },
    ],
  },
];

<AppSwitcher sections={sections} />
```

## Accessibility

`AppSwitcher` is built on Base UI Popover primitives (`@base-ui/react/popover`).

### Keyboard navigation

<KeyboardTable
  entries={[
    { key: 'Enter / Space', action: 'Opens or closes the application switcher popover.' },
    { key: 'Tab', action: 'Moves focus into the search input and across module application links.' },
    { key: 'Escape', action: 'Dismisses the open switcher popover and returns focus to the trigger button.' },
  ]}
/>

### ARIA attributes

- The trigger button receives `aria-haspopup="dialog"` and `aria-expanded="true | false"`.
- Application tiles use semantic anchor links (`<a>`) with accessible titles and descriptions.
- The built-in search filter uses `role="searchbox"` to filter apps dynamically without losing keyboard focus.
