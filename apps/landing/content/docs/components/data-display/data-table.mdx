```tsx

  DataTable,
  type DataTableColumn,
  type DataTableProps,
  type DataTableSlot,
  type SortState,
  type SortValue,
} from '@innostes/ui';
```

## When to use

Use `DataTable` when operators need to browse, filter, sort, select, and manage structured tabular datasets in administrative interfaces:

- Managing core business entities: sales orders, inventory balances, purchase requisitions, and vendor master data.
- Handling paginated server records with sorting, pagination, and bulk batch actions.
- Displaying tabular data that transforms into responsive stacked cards on mobile devices.

When deciding between table implementations:
- Use [Table](/ui/docs/components/data-display/table) when rendering static, short tabular data (such as invoice line item breakdowns or tax summary blocks) where column sorting, multi-selection, and pagination controls are unnecessary.
- Use `DataTable` for entity listings, audit logs, and search results requiring pagination, multi-row selection, or dynamic sorting.
- For extremely large datasets with more than 1,000 simultaneous unpaginated rows rendered in the DOM, combine `Table` with virtual scrolling (`@tanstack/react-virtual`).

## Examples

### Typed columns and accessor mapping

Columns are generic over your row type `TRow`. Define `id`, `header`, `accessor` (used for sorting and default text extraction), and optional custom `cell` render functions.

```tsx
import { Badge, DataTable, type DataTableColumn } from '@innostes/ui';

interface PurchaseOrder {
  id: string;
  vendor: string;
  status: 'Draft' | 'Approved' | 'Dispatched';
  totalAmount: number;
}

const columns: DataTableColumn<PurchaseOrder>[] = [
  {
    id: 'id',
    header: 'PO number',
    accessor: (row) => row.id,
    sortable: true,
    cell: (row) => <span className="font-mono font-medium">{row.id}</span>,
  },
  {
    id: 'vendor',
    header: 'Vendor name',
    accessor: (row) => row.vendor,
    sortable: true,
  },
  {
    id: 'status',
    header: 'Status',
    accessor: (row) => row.status,
    cell: (row) => (
      <Badge variant={row.status === 'Approved' ? 'success' : 'secondary'}>
        {row.status}
      </Badge>
    ),
  },
  {
    id: 'totalAmount',
    header: 'Amount',
    accessor: (row) => row.totalAmount,
    sortable: true,
    align: 'end',
    cell: (row) => (
      <span className="font-mono">
        ₹{row.totalAmount.toLocaleString('en-IN')}
      </span>
    ),
  },
];
```

### Number and currency alignment

Numerical quantities, percentages, and monetary values must always align to the right (`align="end"`). This ensures decimal points and magnitude digits align vertically for rapid visual scanning by accounting operators.

```tsx
const columns: DataTableColumn<InvoiceItem>[] = [
  {
    id: 'quantity',
    header: 'Quantity',
    accessor: (r) => r.quantity,
    align: 'end',
    cell: (r) => <span className="tabular-nums">{r.quantity}</span>,
  },
  {
    id: 'unitPrice',
    header: 'Unit price',
    accessor: (r) => r.unitPrice,
    align: 'end',
    cell: (r) => (
      <span className="tabular-nums">₹{r.unitPrice.toFixed(2)}</span>
    ),
  },
];
```

### Server-side pagination and sorting

When operating on database datasets with thousands of records, set `manualSorting={true}` and `manualPagination={true}`. Pass the current slice of data as `data`, the overall record count as `totalCount`, and handle state transitions through `onSortChange` and `onPageChange`.

<ComponentPreview name="data-display/data-table/server-side" align="start" />

```tsx
'use client';

import { useState, useEffect } from 'react';
import { DataTable, type SortState } from '@innostes/ui';

export function ServerOrdersTable() {
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(true);
  const [page, setPage] = useState(1);
  const [sort, setSort] = useState<SortState | null>({
    columnId: 'date',
    direction: 'desc',
  });
  const [totalCount, setTotalCount] = useState(0);

  useEffect(() => {
    let cancelled = false;
    setLoading(true);

    fetch(`/api/v1/orders?page=${page}&sort=${sort?.columnId}&dir=${sort?.direction}`)
      .then((res) => res.json())
      .then((json) => {
        if (!cancelled) {
          setData(json.items);
          setTotalCount(json.total);
          setLoading(false);
        }
      });

    return () => {
      cancelled = true;
    };
  }, [page, sort]);

  return (
    <DataTable
      aria-label="Server purchase orders"
      data={data}
      columns={columns}
      getRowId={(row) => row.id}
      loading={loading}
      page={page}
      onPageChange={setPage}
      pageSize={10}
      manualPagination
      totalCount={totalCount}
      sort={sort}
      onSortChange={setSort}
      manualSorting
    />
  );
}
```

### Selection and bulk actions

Enable multi-row checkbox selection by setting `selectable={true}`. Pass `selectedIds` and `onSelectedIdsChange` to control the active set of selected IDs.

```tsx
'use client';

import { useState } from 'react';
import { Button, DataTable, toast } from '@innostes/ui';

export function BulkDispatchTable({ orders, columns }) {
  const [selectedIds, setSelectedIds] = useState<string[]>([]);

  const handleBulkDispatch = () => {
    toast.success(`Dispatched ${selectedIds.length} orders successfully.`);
    setSelectedIds([]);
  };

  return (
    <div className="flex flex-col gap-4">
      <div className="flex justify-between items-center">
        <span className="text-sm text-muted-foreground">
          {selectedIds.length} orders selected
        </span>
        <Button
          size="sm"
          disabled={selectedIds.length === 0}
          onClick={handleBulkDispatch}
        >
          Dispatch selected
        </Button>
      </div>

      <DataTable
        aria-label="Orders awaiting dispatch"
        data={orders}
        columns={columns}
        getRowId={(row) => row.id}
        selectable
        selectedIds={selectedIds}
        onSelectedIdsChange={setSelectedIds}
      />
    </div>
  );
}
```

### Loading and empty states

Supply custom `emptyState` messages or let `DataTable` display animated skeleton rows when `loading={true}`.


### Responsive stack mode

On narrow mobile displays or tight dashboard containers, set `responsive="stack"`. The table uses container queries (`@container`) to automatically transform horizontal table rows into card-based key-value pairs without requiring viewport breakpoint hacks.

```tsx
<DataTable
  aria-label="Mobile inventory list"
  data={stockItems}
  columns={columns}
  getRowId={(r) => r.id}
  responsive="stack"
/>
```

## Accessibility

`DataTable` adheres to W3C WAI-ARIA Table and Grid accessibility standards.

### Screen readers and announcements

- Real semantic HTML `<table>`, `<thead>`, `<tbody>`, `<th>`, and `<td>` elements ensure assistive technologies correctly identify row and column counts.
- Column headers with sorting enabled render an embedded `<button>` element with `aria-sort="ascending"`, `aria-sort="descending"`, or `aria-sort="none"`.
- Row selection checkboxes carry localized accessible labels (e.g. "Select row PO-8410").
- The master checkbox uses `aria-label="Select all rows on this page"` and reflects indeterminate state when a subset of page rows is selected.
- A live region announces changes in selection counts politely.
